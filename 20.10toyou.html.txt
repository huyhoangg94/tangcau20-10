        <p id="message">Chúc cậu có một ngày thật nhiều niềm vui, luôn luôn hạnh phúc và gặp nhiều may mắn nhé! ✨</p>
        <button class="btn" onclick="changeMessage()">Bấm vào đây nè ❤️</button>

    <script>
        const messages = [
            "Chúc cậu luôn xinh đẹp, rạng rỡ và tươi cười mỗi ngày! 😊",
            "Cảm ơn vì đã luôn ở đó và mang lại năng lượng tích cực nhé! 🌸",
            "Mong mọi điều tốt đẹp nhất và ngọt ngào nhất sẽ đến với cậu! 🌟",
            "Hôm nay hãy cười thật tươi nha! Chúc mừng ngày đặc biệt! 🎉"
        ];
        let index = 0;
        function changeMessage() {
            const msgElem = document.getElementById("message");
            msgElem.style.opacity = 0;
            setTimeout(() => {
                msgElem.innerText = messages[index];
                msgElem.style.opacity = 1;
                index = (index + 1) % messages.length;
            }, 300);
        }
    </script>
